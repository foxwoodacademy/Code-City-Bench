# modules/document_ingestion.py

import os
import tempfile
import requests
from tika import parser
import logging
import spacy # Import spaCy

class DocumentIngestionPipeline:
    """Document ingestion and entity extraction using Apache Tika."""
    
    def __init__(self, storage_dir=None, spacy_model="en_core_web_sm"):
        """Initialize the document ingestion pipeline.
        
        Args:
            storage_dir (str, optional): Directory to store processed documents.
                If None, a temporary directory will be used.
            spacy_model (str): Name of the spaCy model to load for entity extraction.
        """
        self.storage_dir = storage_dir or tempfile.mkdtemp()
        os.makedirs(self.storage_dir, exist_ok=True)
        logging.info(f"Document storage directory: {self.storage_dir}")
        
        # Initialize spaCy for entity extraction
        try:
            self.nlp = spacy.load(spacy_model)
            logging.info(f"Loaded spaCy model: {spacy_model}")
        except Exception as e:
            logging.error(f"Failed to load spaCy model {spacy_model}: {e}. Ensure it's installed (e.g., 'python -m spacy download en_core_web_sm').")
            self.nlp = None # Fallback if spaCy model cannot be loaded

    def ingest_document(self, file_path=None, file_content=None, url=None):
        """Ingest a document from a file path, content, or URL.
        
        Args:
            file_path (str, optional): Path to the document file.
            file_content (bytes, optional): Raw content of the document.
            url (str, optional): URL to download the document from.
            
        Returns:
            dict: Extracted content and metadata, including extracted entities.
        """
        if sum(1 for x in [file_path, file_content, url] if x is not None) != 1:
            raise ValueError("Exactly one of file_path, file_content, or url must be provided")
        
        parsed_content = None
        
        if url:
            try:
                response = requests.get(url)
                response.raise_for_status()
                parsed_content = parser.from_buffer(response.content)
                logging.info(f"Successfully ingested document from URL: {url}")
            except requests.exceptions.RequestException as e:
                logging.error(f"Error downloading document from URL {url}: {e}")
                raise
        elif file_path:
            if not os.path.exists(file_path):
                raise FileNotFoundError(f"Document file not found: {file_path}")
            try:
                parsed_content = parser.from_file(file_path)
                logging.info(f"Successfully ingested document from file: {file_path}")
            except Exception as e:
                logging.error(f"Error parsing document from file {file_path}: {e}")
                raise
        elif file_content:
            try:
                parsed_content = parser.from_buffer(file_content)
                logging.info("Successfully ingested document from content buffer.")
            except Exception as e:
                logging.error(f"Error parsing document from content buffer: {e}")
                raise

        # Extract entities using the new spaCy-based method
        extracted_entities = []
        if parsed_content and parsed_content.get('content'):
            extracted_entities = self._extract_entities_with_spacy(parsed_content['content'])
        
        return {
            "content": parsed_content.get('content', '').strip() if parsed_content else "",
            "metadata": parsed_content.get('metadata', {}) if parsed_content else {},
            "entities": extracted_entities # Add extracted entities to the output
        }
    
    def _extract_entities_with_spacy(self, content):
        """Extract entities from document content using spaCy.
        
        This method replaces the previous placeholder logic with actual NLP processing.
        It uses a loaded spaCy model to perform Named Entity Recognition (NER).
        
        Args:
            content (str): The text content of the document.
            
        Returns:
            list: A list of dictionaries, where each dictionary represents an entity
                  and contains its 'text' and 'label' (entity type).
                  Example: [{"text": "John Doe", "label": "PERSON"}, {"text": "FBI", "label": "ORG"}]
        """
        entities = []
        if self.nlp and content:
            doc = self.nlp(content)
            for ent in doc.ents:
                # Filter out entities that might not be relevant or are too short
                if len(ent.text.strip()) > 1: # Basic filter for very short entities
                    entities.append({"text": ent.text, "label": ent.label_})
        return entities

    def get_supported_extensions(self):
        """Returns a list of supported document extensions for ingestion."""
        return [
            ".pdf", ".doc", ".docx", ".ppt", ".pptx", ".xls", ".xlsx",
            ".txt", ".rtf", ".html", ".htm", ".xml", ".odt", ".ods", ".odp"
        ]

# modules/document_ingestion.py

import os
import tempfile
import requests
from tika import parser
import logging

logger = logging.getLogger("DocumentIngestionPipeline")

class DocumentIngestionPipeline:
    """Document ingestion and content extraction using Apache Tika."""
    
    def __init__(self, storage_dir=None):
        """Initialize the document ingestion pipeline.
        
        Args:
            storage_dir (str, optional): Directory to store processed documents.
                If None, a temporary directory will be used.
        """
        self.storage_dir = storage_dir or tempfile.mkdtemp()
        os.makedirs(self.storage_dir, exist_ok=True)
        logger.info(f"Document storage directory: {self.storage_dir}")
    
    def _download_from_url(self, url: str, temp_filepath: str) -> bool:
        """
        Downloads a document from a URL to a temporary file.
        
        Args:
            url (str): The URL of the document.
            temp_filepath (str): The path where the downloaded file will be saved.
            
        Returns:
            bool: True if download was successful, False otherwise.
        """
        try:
            logger.info(f"Attempting to download document from URL: {url}")
            response = requests.get(url, stream=True, timeout=30)
            response.raise_for_status() # Raise an HTTPError for bad responses (4xx or 5xx)
            
            with open(temp_filepath, 'wb') as f:
                for chunk in response.iter_content(chunk_size=8192):
                    f.write(chunk)
            logger.info(f"Successfully downloaded document to {temp_filepath}")
            return True
        except requests.exceptions.RequestException as e:
            logger.error(f"Error downloading document from {url}: {e}")
            return False
        except Exception as e:
            logger.error(f"An unexpected error occurred during download from {url}: {e}", exc_info=True)
            return False

    def _parse_document_content(self, filepath: str) -> dict:
        """
        Parses document content and metadata using Apache Tika.
        
        Args:
            filepath (str): The path to the document file.
            
        Returns:
            dict: Parsed content and metadata.
        """
        try:
            logger.info(f"Parsing document content using Tika for file: {filepath}")
            parsed = parser.from_file(filepath)
            content = parsed['content'] if parsed and 'content' in parsed else ""
            metadata = parsed['metadata'] if parsed and 'metadata' in parsed else {}
            logger.info(f"Tika parsing successful for {filepath}")
            return {"content": content, "metadata": metadata}
        except Exception as e:
            logger.error(f"Error parsing document with Tika from {filepath}: {e}", exc_info=True)
            return {"content": "", "metadata": {"error": str(e)}}

    def ingest_document(self, file_path: str = None, file_content: bytes = None, url: str = None) -> dict:
        """
        Ingest a document from a file path, content, or URL, and return its parsed content.
        
        Args:
            file_path (str, optional): Path to the document file.
            file_content (bytes, optional): Raw content of the document.
            url (str, optional): URL to download the document from.
            
        Returns:
            dict: Extracted content and metadata, or an error dictionary.
        """
        if sum(1 for x in [file_path, file_content, url] if x is not None) != 1:
            logger.error("Exactly one of file_path, file_content, or url must be provided for ingestion.")
            return {"content": "", "metadata": {"error": "Invalid input: Provide exactly one source."}}
        
        temp_file = None
        try:
            if url:
                # Create a temporary file for download
                suffix = "." + url.split('.')[-1] if '.' in url else ""
                temp_file = tempfile.NamedTemporaryFile(delete=False, suffix=suffix, dir=self.storage_dir)
                temp_filepath = temp_file.name
                temp_file.close() # Close the handle so other processes (like Tika) can open it
                
                if not self._download_from_url(url, temp_filepath):
                    return {"content": "", "metadata": {"error": f"Failed to download document from {url}"}}
                
                return self._parse_document_content(temp_filepath)
            
            elif file_content:
                # Write content to a temp file for Tika processing
                temp_file = tempfile.NamedTemporaryFile(delete=False, dir=self.storage_dir)
                temp_file.write(file_content)
                temp_filepath = temp_file.name
                temp_file.close() # Close the handle
                return self._parse_document_content(temp_filepath)
                
            elif file_path:
                # Use existing file path
                if not os.path.exists(file_path):
                    logger.error(f"File not found: {file_path}")
                    return {"content": "", "metadata": {"error": f"File not found: {file_path}"}}
                return self._parse_document_content(file_path)
            
        except Exception as e:
            logger.error(f"Error during document ingestion: {e}", exc_info=True)
            return {"content": "", "metadata": {"error": f"Document ingestion failed: {e}"}}
        finally:
            if temp_file and os.path.exists(temp_filepath):
                os.unlink(temp_filepath) # Clean up temp file
                logger.debug(f"Cleaned up temporary file: {temp_filepath}")

    def get_supported_extensions(self) -> list:
        """
        Returns a list of supported document extensions for ingestion.
        """
        return [
            ".pdf", ".doc", ".docx", ".ppt", ".pptx", ".xls", ".xlsx",
            ".txt", ".rtf", ".html", ".htm", ".xml", ".odt", ".ods", ".odp"
        ]

                            event_date = datetime.strptime(date_match.group(0).replace("on ", "").replace("in ", ""), '%b %d, %Y').strftime('%Y-%m-%d')
                            mock_events.append({"date": event_date, "description": f"Red flag '{flag_text}' detected."})
                        except ValueError:
                            pass # Date parsing failed
        
        if mock_events:
            # Sort events chronologically
            insight["events"] = sorted(mock_events, key=lambda x: x["date"])
            insight["summary"] = f"Identified {len(insight['events'])} key events on the timeline."
            
        logger.info("Timeline analysis insights generated.")
        return insight

    def _generate_risk_assessment_insights(self, documents: List[str], entities: List[Dict[str, str]], red_flags: Dict[str, Any], mode: str) -> Dict[str, Any]:
        """
        Assesses overall risk based on red flags and entity connections.
        
        Args:
            documents (list): List of document contents.
            entities (list): List of extracted entities.
            red_flags (dict): Red flag analysis report.
            mode (str): Current analysis mode.
            
        Returns:
            dict: Insights related to risk assessment.
        """
        insight = {
            "type": "risk_assessment",
            "summary": "Overall risk level: Low.",
            "risk_score": 0.0, # 0-100 scale
            "risk_categories": []
        }
        
        total_severity = 0
        num_categories = 0
        risk_categories_detail = []

        if red_flags and red_flags.get("severity_scores"):
            for category, score in red_flags["severity_scores"].items():
                total_severity += score
                num_categories += 1
                risk_categories_detail.append({
                    "category": category.replace('_', ' ').title(),
                    "score": score,
                    "description": f"Severity {score}/10 based on {len(red_flags['red_flags'].get(category, []))} instances."
                })
        
        # Adjust risk based on mode (e.g., "audit" or "psychological_warfare" might imply higher base risk)
        mode_multiplier = 1.0
        if mode == "audit":
            mode_multiplier = 1.2
        elif mode == "psychological_warfare":
            mode_multiplier = 1.5

        if num_categories > 0:
            avg_severity_scaled = (total_severity / num_categories) * 10 * mode_multiplier # Scale to 0-100
            insight["risk_score"] = round(min(100.0, avg_severity_scaled), 1) # Cap at 100
        
        if insight["risk_score"] >= 70:
            insight["summary"] = "Overall risk level: Critical. Immediate action required."
        elif insight["risk_score"] >= 40:
            insight["summary"] = "Overall risk level: Moderate. Further investigation recommended."
        else:
            insight["summary"] = "Overall risk level: Low. Monitoring advised."
            
        insight["risk_categories"] = risk_categories_detail
            
        logger.info("Risk assessment insights generated.")
        return insight

    def generate_insights(self, documents: List[str], entities: List[Dict[str, str]], red_flags: Dict[str, Any], mode: str = "standard") -> List[Dict[str, Any]]:
        """
        Generates various types of insights based on the provided data.
        
        Args:
            documents (list): List of document contents (raw text).
            entities (list): List of extracted entities (e.g., [{'name': 'John Doe', 'type': 'PERSON'}]).
            red_flags (dict): Red flag analysis report from EntityParsingRedFlagAnalyzer.
            mode (str): The analysis mode (e.g., "standard", "deep", "psychological_warfare").
            
        Returns:
            list: A list of generated insight dictionaries.
        """

            generator_func = self.insight_generators.get(generator_name)
            if generator_func:
                try:
                    insight_data = generator_func(documents, entities, red_flags, mode)
                    self.insights.append(insight_data)
                    logger.info(f"Generated insight: {generator_name}")
                except Exception as e:
                    logger.error(f"Error generating insight '{generator_name}': {e}", exc_info=True)
                    self.insights.append({"type": generator_name, "summary": f"Error generating insight: {e}"})
            else:
                logger.warning(f"No generator function found for insight type: {generator_name}")
        
        # Add a meta-insight or overall summary if desired
        overall_summary = self._generate_overall_aura_summary(self.insights)
        self.insights.append({"type": "overall_aura_summary", "summary": overall_summary})

        logger.info(f"Aura insights generation complete. Generated {len(self.insights)} insights.")
        return self.insights

            elif insight["type"] == "timeline_analysis" and insight.get("events"):
                summary_parts.append(f"Mapped {len(insight['events'])} critical events on a timeline.")
            elif insight["type"] == "risk_assessment":

            elif insight["type"] == "strategic_recommendations" and insight.get("recommendations"):
                critical_recs = [r for r in insight["recommendations"] if r.get("priority") == "critical"]

            elif insight["type"] == "psychological_profile" and insight.get("profiles"):
                summary_parts.append(f"Developed psychological profiles for {len(insight['profiles'])} key individuals.")
        
        if summary_parts:
            return " ".join(summary_parts) + "."
        else:
            return "Analysis complete, but no significant patterns or insights were identified in the available data."

Module Role Summary and Interactions
Here's a summary of each module's role and how they interact within the Veritas AI system:

config.py (Configuration Module)

Role: Centralizes application-wide settings such as API keys (GEMINI_API_KEY) and logging configurations (LOGGING_CONFIG).

Interactions:

app.py: Imports GEMINI_API_KEY and LOGGING_CONFIG during application initialization.

modules/services/gemini_service.py: Imports GEMINI_API_KEY to configure the Gemini API client.

All Modules: Indirectly affects all modules through the logging configuration, ensuring consistent log output.

modules/services/gemini_service.py (Gemini Service Module)

Role: Provides a dedicated, encapsulated interface for interacting with the Google Gemini API. Its primary responsibility is to send prompts and receive AI-generated content.

Interactions:

app.py: app.py initializes an instance of GeminiService and calls its generate_content method to get AI responses for scenario analysis. It handles exceptions raised by this service.

config.py: Reads the GEMINI_API_KEY from config.py during its initialization.

modules/document_ingestion.py (Document Ingestion Module)

Role: Handles the ingestion of documents from various sources (URLs, local files, raw content) and extracts their raw text and metadata using Apache Tika. It is solely responsible for getting document content into a usable format.

Interactions:

app.py: app.py initializes DocumentIngestionPipeline and calls its ingest_document method when a document URL or file content is provided in the analysis request. The extracted content is then passed to entity_analyzer.

External (Apache Tika): Directly interacts with the Apache Tika server (assumed to be running externally) for parsing diverse document formats.

External (Web): Makes HTTP requests to download documents from provided URLs.

modules/entity_analyzer.py (Entity Analyzer Module)

Role: Performs Named Entity Recognition (NER) on text content using spaCy (with a fallback if spaCy fails) and applies predefined regex-based rules to detect "red flags" indicative of misconduct. It also calculates severity scores and generates a summary of these findings.

Interactions:

app.py: app.py initializes EntityParsingRedFlagAnalyzer. It calls extract_entities_from_content to get entities from ingested documents/scenario text, and then analyze_entities_for_red_flags to generate the red flag report. The results are then passed to app.py and subsequently to aura_agent.

modules/document_ingestion.py: Receives raw text content from document_ingestion.py for processing.

External (spaCy): Depends on a pre-trained spaCy NLP model for entity extraction.

modules/foia_generator.py (FOIA Generator Module)

Role: Automates the creation of standard FOIA requests and appeal letters. It manages templates and agency contact information, and can analyze denial letters to formulate appeal arguments.

Interactions:

# modules/report_generator.py



import os

import datetime

from jinja2 import Environment, FileSystemLoader

from weasyprint import HTML



OUTPUT_DIR = "output/reports"



def generate_pdf_report(analysis_data, template_name="report_template.html"):

"""Generate a PDF report from analysis data."""

os.makedirs(OUTPUT_DIR, exist_ok=True)



env = Environment(loader=FileSystemLoader("templates"))

template = env.get_template(template_name)



html_content = template.render(data=analysis_data)

timestamp = datetime.datetime.now().strftime("%Y%m%d-%H%M%S")

filename = f"{timestamp}-veritas-report.pdf"

output_path = os.path.join(OUTPUT_DIR, filename)



HTML(string=html_content).write_pdf(output_path)



return output_path
