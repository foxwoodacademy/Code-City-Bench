    <script>
        const scenarioInput = document.getElementById('scenarioInput');
        const analyzeBtn = document.getElementById('analyzeBtn');
        const outputArea = document.getElementById('outputArea');
        const loader = document.getElementById('loader');
        const exposedOverlay = document.getElementById('exposedOverlay');
        const puddleEffect = document.getElementById('puddleEffect');

        // Function to simulate AI analysis and response
        async function simulateAnalysis() {
            const input = scenarioInput.value.trim();
            if (!input) {
                outputArea.innerHTML = '<p class="text-red-400 text-center">Please describe a scenario to analyze.</p>';
                return;
            }

            // Show loading state
            analyzeBtn.disabled = true;
            loader.classList.remove('hidden');
            analyzeBtn.querySelector('span:last-child').classList.add('hidden');
            outputArea.innerHTML = '<p class="text-blue-400 text-center animate-pulse">Analyzing patterns... Identifying vulnerabilities...</p>';
            exposedOverlay.classList.remove('active');
            puddleEffect.classList.remove('active');

            try {
                // Simulate network delay
                await new Promise(resolve => setTimeout(resolve, 2000));

                // --- LLM Call (Gemini API) ---
                const prompt = `Given the following alleged wrongdoing or corrupt scenario, provide a direct, uncompromising, and legally-oriented analysis of potential ethical breaches, legal ramifications, and the concept of accountability. Focus on the impact on the wrongdoer and the system, using strong, formal language. Do not provide legal advice, but highlight the principles of justice.

                Scenario: "${input}"

                Coach Bleak's Analysis:`;

                let chatHistory = [];
                chatHistory.push({ role: "user", parts: [{ text: prompt }] });

                const payload = { contents: chatHistory };
                const apiKey = ""; // Canvas will provide this key at runtime
                const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.0-flash:generateContent?key=${apiKey}`;

                const response = await fetch(apiUrl, {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify(payload)
                });

                if (!response.ok) {
                    const errorData = await response.json();
                    throw new Error(`API error: ${errorData.error.message || response.statusText}`);
                }

                const result = await response.json();

                let aiResponse = '';
                if (result.candidates && result.candidates.length > 0 &&
                    result.candidates[0].content && result.candidates[0].content.parts &&
                    result.candidates[0].content.parts.length > 0) {
                    aiResponse = result.candidates[0].content.parts[0].text;
                } else {
                    aiResponse = "Analysis inconclusive. Further data required. The truth remains obscured for now.";
                }

                // Display analysis
                outputArea.innerHTML = `<p class="text-gray-200">${aiResponse.replace(/\n/g, '<br>')}</p>`;

                // Trigger "exposed" visuals
                exposedOverlay.classList.add('active');
                puddleEffect.classList.add('active');

            } catch (error) {
                console.error("Coach Bleak encountered an error:", error);
                outputArea.innerHTML = `<p class="text-red-400">ERROR: Analysis failed. ${error.message}. Ensure valid input and network connection.</p>`;
            } finally {
                // Reset loading state
                analyzeBtn.disabled = false;
                loader.classList.add('hidden');
                analyzeBtn.querySelector('span:last-child').classList.remove('hidden');
            }
        }

        analyzeBtn.addEventListener('click', simulateAnalysis);

        // Optional: Reset visuals when input changes
        scenarioInput.addEventListener('input', () => {
            exposedOverlay.classList.remove('active');
            puddleEffect.classList.remove('active');
            outputArea.innerHTML = '<p class="text-gray-400 text-center"><span class="text-blue-400">Coach Bleak</span> awaits your input. Describe the scenario above and click "UNVEIL THE TRUTH" to see the potential ramifications.</p>';
        });
    </script>
