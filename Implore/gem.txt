        echo ""
        echo -e "${CYAN}📖 Usage:${NC}"

        echo "   3. Or use Telegram bot (if configured)"
        echo ""
        echo -e "${BLUE}💡 oclog to view logs | ockill to stop service${NC}"
    else
        echo -e "${RED}❌ Service failed to start, please check logs (openclaw_final logs)${NC}"
    fi
}

oclog() {
    if tmux has-session -t openclaw_final 2>/dev/null; then
        tmux attach -t openclaw_final
    else
        echo -e "${YELLOW}⚠️  OpenClaw service not running, use ocr to start${NC}"
    fi
}

ockill() {
    echo -e "${YELLOW}Stopping OpenClaw service...${NC}"
    pkill -9 -f "openclaw_final" 2>/dev/null
    tmux kill-session -t openclaw_final 2>/dev/null
    sleep 1
    if ! tmux has-session -t openclaw_final 2>/dev/null && ! pgrep -f "openclaw_final" > /dev/null; then
        echo -e "${GREEN}✅ OpenClaw service stopped${NC}"
    else
        echo -e "${RED}❌ Failed to stop service, please check manually${NC}"
    fi
}
