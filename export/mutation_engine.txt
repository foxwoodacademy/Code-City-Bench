class State:
    def __init__(self):
        self.data = {
            "last_run": None,
            "invoices_created": 0,
            "payments_settled": 0,
            "copy_generated": 0,
            "leads_contacted": 0,

        if DASH_STATE.exists():

        with open(DASH_STATE, 'w') as f:
            json.dump(self.data, f, indent=2)
    
    def log(self, msg, level="INFO"):
        entry = f"[{datetime.now().isoformat()}] {level}: {msg}"
        self.data["errors"].append(entry)
        self.data["errors"] = self.data["errors"][-50:]  # Keep last 50
        with open(LOG_FILE, 'a') as f:
            f.write(entry + "\n")
    
    def trigger(self, event, payload=None):
        self.data["trigger_queue"].append({
            "ts": datetime.now().isoformat(),
            "event": event,
            "payload": payload
        })
        self.save()

            self.scr.addstr(3 + len(self.menu_items) + 2 + i, 2, line[:w-3])
        
        # Action required
        y = 3 + len(self.menu_items) + 2 + 5
        self.scr.addstr(y, 2, "Actions Required:", curses.A_BOLD)
        # (Would populate from briefing)
        
        self.scr.refresh()
    
    def handle_input(self):
        try:
            key = self.scr.getch()
            if key == -1:
                return
            
            if key == curses.KEY_UP:
                self.selected = max(0, self.selected - 1)
            elif key == curses.KEY_DOWN:
                self.selected = min(len(self.menu_items) - 1, self.selected + 1)
            elif key == ord('\n'):
                self.execute()
            elif key == ord('q'):
                self.running = False
        except:
            pass
    
    def execute(self):
        item = self.menu_items[self.selected]
        STATE.log(f"Executing: {item}")
        
        if item == "RUN FULL PIPELINE":
            STATE.data["mode"] = "SHIP"
            MrsHiggins.run_pipeline()
            STATE.data["mode"] = "IDLE"
        
        elif item == "GENERATE LAUNCH KIT":

            MrsHiggins.check_payments()
        
        elif item == "RUN SCOUT":
            VerticalAI.run_scout("askhole")
        
        elif item == "DAILY BRIEFING":
            AskHoleAuto.daily_briefing()
        
        elif item == "TOGGLE MONITOR MODE":
            STATE.data["mode"] = "MONITOR" if STATE.data["mode"] != "MONITOR" else "IDLE"
        
        elif item == "VIEW LOGS":
            # Switch to log viewer mode (simplified)
            pass
        
        elif item == "PANIC RESET":
            STATE.data["mode"] = "PANIC"
            STATE.data["trigger_queue"] = []
            STATE.data["errors"] = []
            STATE.save()
            STATE.data["mode"] = "IDLE"
        
        elif item == "QUIT":
            self.running = False
        
        STATE.save()
    
    def run(self):

            self.draw()

            time.sleep(0.1)
