/**
 * PHASE 2: Manual AI Explanation Generation
 * In the MVP, this function will prompt the user to paste the code block
 * into the Gemini chat, and the user will manually copy the resulting 
 * structured HTML back into a temp file for display.
 * This function handles the UI instructions for that manual process.
 */
export async function callAIForExplanation(code: string): Promise<string> {
    const tutorialPrompt = `
    Please copy the code block below and paste it into a new Gemini chat
    with the following instruction.
    
    --- START GEMINI INSTRUCTION ---
    You are an expert Code Tutor and VS Code extension AI. Your goal is to explain the provided block of code using a structured, tutorial format suitable for a beginner or intermediate developer. The output must be a single string of HTML content, ready to be injected directly into a VS Code webview panel, using the following exact structure and rules:
    
    **Structure (Must use these three H3 headings in order):**
    1.  ### Overview and Purpose
    2.  ### Step-by-Step Breakdown
    3.  ### Mentorship Challenge
    
    **Input Code:**
    
    \`\`\`
    ${code}
    \`\`\`
    
    **Format Rules:**
    * Use standard **HTML** tags (<h1>, <h3>, <p>, <ul>, <ol>, <li>, <code>) only. Do not use Markdown.
    * The entire output must be one single string.
    
    --- END GEMINI INSTRUCTION ---
    
    `;

    // The explanation shown in the Webview panel *before* the user pastes the final HTML
    const guidanceHTML = `
    <!DOCTYPE html>
    <html lang="en">
    <body>
        <h1>CodeMentor Manual Step 🧠</h1>
        <p>Your code block has been captured!</p>
        
        <h2>ACTION REQUIRED:</h2>
        <ol>
            <li><strong>Copy the Code Block</strong> shown below.</li>
            <li>Paste the Code Block and the instruction above into our chat.</li>
            <li><strong>Copy the full HTML output</strong> that Gemini generates for you.</li>
            <li>Return here and paste that HTML into a file named <code>/tmp/mentor_output.html</code> (or similar) on your system.</li>
        </ol>

        <h3>Captured Code Block:</h3>
        <pre>${code}</pre>
        
        <p><em>(In a future version, this step will be automated using the Gemini API key.)</em></p>
    </body>
    </html>`;
    
    // In a real execution, we would resolve this with the *final* HTML content.
    // For now, we return the instruction HTML so the user can proceed.
    return Promise.resolve(guidanceHTML);
}

import * as vscode from 'vscode';
import { callAIForExplanation } from './callAIForExplanation';

let currentPanel: vscode.WebviewPanel | undefined = undefined;

export function activate(context: vscode.ExtensionContext) {

	let disposable = vscode.commands.registerCommand('codementor.explainLastBlock', async () => {
		const editor = vscode.window.activeTextEditor;

		if (!editor) {
			vscode.window.showInformationMessage('No active editor open.');
			return;
		}

		const selection = editor.selection;
		const codeBlock = editor.document.getText(selection);

		if (codeBlock.length === 0) {
			vscode.window.showInformationMessage('No code selected to explain.');
			return;
		}

		// 1. Get Explanation from the Placeholder Function
		// This is where Phase 2 (Gemini Advanced) will be integrated
		const explanation = await callAIForExplanation(codeBlock);

		// 2. Create/Show Webview Panel
		if (currentPanel) {
			currentPanel.reveal(vscode.ViewColumn.One);
		} else {
			currentPanel = vscode.window.createWebviewPanel(
				'codeMentorView',
				'CodeMentor',
				vscode.ViewColumn.Two,
				{
					enableScripts: true
				}
			);

			currentPanel.onDidDispose(
				() => {
					currentPanel = undefined;
				},
				null,
				context.subscriptions
			);
		}

		// 3. Set Webview Content
		currentPanel.webview.html = getWebviewContent(codeBlock, explanation);
	});

	context.subscriptions.push(disposable);
}

function getWebviewContent(code: string, explanation: string): string {
	// A simple HTML structure for the Webview
	return `<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>CodeMentor</title>
    <style>
        body { font-family: sans-serif; padding: 15px; }
        pre { background-color: #333; color: #f0f0f0; padding: 10px; border-radius: 5px; overflow-x: auto; }
        h3 { color: #88B04B; border-bottom: 2px solid #88B04B; padding-bottom: 5px; }
    </style>
</head>
<body>
    <h1>CodeMentor Explanation</h1>
    
    <h3>Selected Code Block:</h3>
    <pre>${code}</pre>

    <h3>AI Explanation:</h3>
    <p>${explanation}</p>
</body>
</html>`;
}

export function deactivate() {}

# --- USAGE TRACKING (for SaaS billing later) ---
log_usage() {
    local cmd="$1" file="$2"
    local logfile="$ANALYTICS/usage_$(date +%Y%m).jsonl"
    echo "{\"ts\":\"$(date -Iseconds)\",\"cmd\":\"$cmd\",\"file\":\"$file\",\"device\":\"$(uname -a | base64 | head -c20)\"}" >> "$logfile"
}

setup() {
    echo -e "${CYAN}╔══════════════════════════════════════╗${NC}"
    echo -e "${CYAN}║     CodeMentor SaaS Setup v2.0       ║${NC}"
    echo -e "${CYAN}╚══════════════════════════════════════╝${NC}"
    echo ""
    echo "Select AI Provider:"
    echo "  [1] Gemini (free tier: 1500 req/day)"
    echo "  [2] OpenRouter (pay-as-you-go, 100+ models)"
    echo ""
    echo -n "Choice [1/2]: "
    read -r choice
    
    if [[ "$choice" == "2" ]]; then
        echo -n "OpenRouter API Key: "
        read -r key
        echo -n "Model [anthropic/claude-3.5-sonnet]: "
        read -r model
        model=${model:-anthropic/claude-3.5-sonnet}
        echo "{\"provider\":\"openrouter\",\"api_key\":\"$key\",\"model\":\"$model\"}" > "$CONFIG"
    else
        echo -n "Gemini API Key: "

    python3 -c "
import http.server, socketserver, os, json, glob
from datetime import datetime

BASE = '$BASE'
ANALYTICS = BASE + '/analytics'

class Handler(http.server.SimpleHTTPRequestHandler):
    def do_GET(self):

<style>
body {{ background:#0f0f1e; color:#e0e0e0; font-family:monospace; padding:20px; }}

<p style=\"margin-top:30px;color:#888;\">Generated: {datetime.now().isoformat()}</p>

            self.wfile.write(html.encode())

            self.send_response(200)
            self.send_header('Content-type', 'application/json')

def analyze_code(code, filepath="unknown"):
    """Static analysis + metrics generation (no external deps required)"""
    lines = code.split('\n')
    metrics = {
        "timestamp": datetime.now().isoformat(),
        "filepath": filepath,
        "total_lines": len(lines),
        "code_lines": len([l for l in lines if l.strip() and not l.strip().startswith('#')]),
        "comment_lines": len([l for l in lines if l.strip().startswith('#')]),
        "blank_lines": len([l for l in lines if not l.strip()]),
        "functions": [],
        "classes": [],
        "imports": [],
        "complexity_estimate": 0,
        "security_flags": []
    }
    
    try:
        tree = ast.parse(code)
        for node in ast.walk(tree):
            if isinstance(node, ast.FunctionDef):
                metrics["functions"].append({
                    "name": node.name,
                    "args": len(node.args.args),
                    "line": node.lineno,
                    "complexity": len(list(ast.walk(node)))
                })
                metrics["complexity_estimate"] += len(list(ast.walk(node)))
            elif isinstance(node, ast.ClassDef):
                metrics["classes"].append({
                    "name": node.name,
                    "methods": len([n for n in ast.walk(node) if isinstance(n, ast.FunctionDef)]),
                    "line": node.lineno
                })
            elif isinstance(node, (ast.Import, ast.ImportFrom)):
                for alias in node.names:
                    metrics["imports"].append(alias.name)
    except SyntaxError:
        metrics["parse_error"] = True
    
    # Security heuristics
    risky = ['eval(', 'exec(', 'subprocess.call', 'os.system(', '__import__', 'input(', 'pickle.loads']
    for risk in risky:
        if risk in code:
            metrics["security_flags"].append(f"Potential risk: {risk}")
    
    return metrics

def call_gemini(code, api_key, model="gemini-2.0-flash"):
    url = f"https://generativelanguage.googleapis.com/v1beta/models/{model}:generateContent?key={api_key}"
    prompt = f"""You are an expert Code Tutor and Security Analyst. Explain this code using ONLY HTML tags.
Structure:
<h3>Overview and Purpose</h3>
<h3>Step-by-Step Breakdown</h3>  
<h3>Security & Best Practices</h3>
<h3>Mentorship Challenge</h3>
Code: ```{code}```"""
    body = json.dumps({"contents":[{"parts":[{"text":prompt}]}]}).encode()
    req = urllib.request.Request(url, data=body, headers={"Content-Type":"application/json"})
    try:
        with urllib.request.urlopen(req, timeout=45) as r:
            resp = json.loads(r.read())
            return resp["candidates"][0]["content"]["parts"][0]["text"]
    except Exception as e:
        return f"<h3>Error</h3><p>API Error: {e}</p><p>Check your API key or network.</p>"

def call_openrouter(code, api_key, model="anthropic/claude-3.5-sonnet"):
    """Fallback to OpenRouter (pay-as-you-go, no Google account needed)"""
    url = "https://openrouter.ai/api/v1/chat/completions"
    headers = {
        "Authorization": f"Bearer {api_key}",
        "Content-Type": "application/json",
        "HTTP-Referer": "https://codementor.local",
        "X-Title": "CodeMentor MVP"
    }
    body = json.dumps({
        "model": model,
        "messages": [{"role":"user","content":f"Explain this code in HTML format with h3 headings: Overview, Breakdown, Security, Challenge. Code: ```{code}```"}]
    }).encode()
    req = urllib.request.Request(url, data=body, headers=headers)
    try:
        with urllib.request.urlopen(req, timeout=45) as r:
            resp = json.loads(r.read())
            return resp["choices"][0]["message"]["content"]
    except Exception as e:
        return f"<h3>Error</h3><p>OpenRouter Error: {e}</p>"
