    <script>
        document.addEventListener('DOMContentLoaded', () => {
            // Card flip logic
            const cards = document.querySelectorAll('.card-flip-container');
            cards.forEach(card => {
                card.addEventListener('click', (e) => {
                    e.preventDefault();
                    card.classList.toggle('flipped');
                });
            });

            // Parallax effect for hero section (subtle)
            const hero = document.getElementById('hero');
            const heroContent = hero.querySelector('.relative.z-20');
            window.addEventListener('scroll', () => {
                const scrollPosition = window.pageYOffset;
                if (scrollPosition < window.innerHeight) {
                    heroContent.style.transform = `translateY(${scrollPosition * 0.2}px)`;
                }
            });

            // --- Brand Ideation Lab Logic ---
            const brandInput = document.getElementById('brandInput');
            const generateSloganBtn = document.getElementById('generateSloganBtn');
            const expandConceptBtn = document.getElementById('expandConceptBtn');
            const refineToneBtn = document.getElementById('refineToneBtn');
            const brandOutput = document.getElementById('brandOutput');
            const brandErrorMessageDiv = document.getElementById('brandErrorMessage');
            const brandErrorTextSpan = document.getElementById('brandErrorText');

            // Helper function to show/hide loading spinner for brand lab buttons
            function setBrandLoading(buttonId, isLoading) {
                const button = document.getElementById(buttonId);
                const loader = button.querySelector('span.animate-spin');
                const textSpan = button.querySelector('span:last-child');

                if (isLoading) {
                    button.disabled = true;
                    loader.classList.remove('hidden');
                    textSpan.classList.add('hidden');
                } else {
                    button.disabled = false;
                    loader.classList.add('hidden');
                    textSpan.classList.remove('hidden');
                }
            }

            // Helper function to display brand lab errors
            function displayBrandError(message) {
                brandErrorTextSpan.textContent = message;
                brandErrorMessageDiv.classList.remove('hidden');
            }

            // Helper function to clear brand lab errors
            function clearBrandError() {
                brandErrorMessageDiv.classList.add('hidden');
                brandErrorTextSpan.textContent = '';
            }

            // Function to call Gemini API for brand ideation
            async function callGeminiBrandAPI(prompt, buttonId) {
                clearBrandError();
                brandOutput.innerHTML = '<p class="text-gray-500">Generating ideas...</p>';

                // Disable all brand lab buttons during API call
                setBrandLoading('generateSloganBtn', true);
                setBrandLoading('expandConceptBtn', true);
                setBrandLoading('refineToneBtn', true);

                // Activate specific button's loader
                setBrandLoading(buttonId, true);

                if (!brandInput.value.trim()) {
                    displayBrandError('Please enter some text in the input field.');
                    // Re-enable all buttons on error
                    setBrandLoading('generateSloganBtn', false);
                    setBrandLoading('expandConceptBtn', false);
                    setBrandLoading('refineToneBtn', false);
                    brandOutput.innerHTML = '<p class="text-gray-500">AI-generated branding ideas will appear here.</p>';
                    return;
                }

                try {
                    let chatHistory = [];
                    chatHistory.push({ role: "user", parts: [{ text: prompt }] });

                    const payload = { contents: chatHistory };
                    const apiKey = ""; // Canvas will provide this key at runtime
                    const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.0-flash:generateContent?key=${apiKey}`;

                    const response = await fetch(apiUrl, {
                        method: 'POST',
                        headers: { 'Content-Type': 'application/json' },
                        body: JSON.stringify(payload)
                    });

                    if (!response.ok) {
                        const errorData = await response.json();
                        throw new Error(`API error: ${errorData.error.message || response.statusText}`);
                    }

                    const result = await response.json();

                    if (result.candidates && result.candidates.length > 0 &&
                        result.candidates[0].content && result.candidates[0].content.parts &&
                        result.candidates[0].content.parts.length > 0) {
                        const text = result.candidates[0].content.parts[0].text;
                        brandOutput.innerHTML = `<p>${text.replace(/\n/g, '<br>')}</p>`;
                    } else {
                        displayBrandError('Could not generate branding ideas. Please try again.');
                        brandOutput.innerHTML = '<p class="text-gray-500">AI-generated branding ideas will appear here.</p>';
                    }
                } catch (err) {
                    console.error('Error calling Gemini API for branding:', err);
                    displayBrandError(`Failed to generate ideas: ${err.message}.`);
                    brandOutput.innerHTML = '<p class="text-gray-500">AI-generated branding ideas will appear here.</p>';
                } finally {
                    // Re-enable all buttons
                    setBrandLoading('generateSloganBtn', false);
                    setBrandLoading('expandConceptBtn', false);
                    setBrandLoading('refineToneBtn', false);
                }
            }

            // Event listeners for Brand Ideation Lab buttons
            generateSloganBtn.addEventListener('click', () => {
                const input = brandInput.value.trim();
                if (input) {
                    const prompt = `Generate 5 concise and impactful brand slogans/taglines for a legal AI named Coach Bleak, based on the following core idea/concept: "${input}". Focus on themes of truth, accountability, precision, and unwavering justice.`;
                    callGeminiBrandAPI(prompt, 'generateSloganBtn');
                } else {
                    displayBrandError('Please enter a brand idea to generate slogans.');
                }
            });

            expandConceptBtn.addEventListener('click', () => {
                const input = brandInput.value.trim();
                if (input) {
                    const prompt = `Expand on the following Coach Bleak brand concept, detailing its visual implications, emotional resonance, and potential applications: "${input}".`;
                    callGeminiBrandAPI(prompt, 'expandConceptBtn');
                } else {
                    displayBrandError('Please enter a concept to expand.');
                }
            });

            refineToneBtn.addEventListener('click', () => {
                const input = brandInput.value.trim();
                if (input) {
                    const prompt = `Refine the tone for the Coach Bleak brand based on the following input: "${input}". Describe how this tone would manifest in communication, visual design, and user interaction.`;
                    callGeminiBrandAPI(prompt, 'refineToneBtn');
                } else {
                    displayBrandError('Please enter a tone or concept to refine.');
                }
            });
        });
    </script>
