                                            }

            // Function to call Gemini API for report interrogation
            async function callGeminiReportInterrogation(prompt, buttonId) {
                clearInterrogationError();
                interrogationOutput.innerHTML = '<p class="text-gray-400">Coach Bleak is processing your query...</p>';

                // Disable all interrogation buttons during API call
                setInterrogationLoading('askReportBtn', true);
                setInterrogationLoading('extendScenarioBtn', true);

                // Activate specific button's loader
                setInterrogationLoading(buttonId, true);

                if (!interrogationInput.value.trim()) {
                    displayInterrogationError('Please enter a query in the input field.');
                    setInterrogationLoading('askReportBtn', false);
                    setInterrogationLoading('extendScenarioBtn', false);
                    interrogationOutput.innerHTML = '<p class="text-gray-400">Coach Bleak is ready for your questions about this audit report.</p>';
                    return;
                }

                try {
                    let chatHistory = [];
                    // Provide the full report content as context
                    chatHistory.push({ role: "user", parts: [{ text: `Based on the following audit report, answer the user's query or extend the hypothetical scenario. Do not introduce new factual information outside of the report, but use its details for analysis and extrapolation.

                    Audit Report:
                    ${fullReportContent}

                    User Query: ${prompt}` }] });

                    const payload = { contents: chatHistory };
                    const apiKey = ""; // Canvas will provide this key at runtime
                    const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.0-flash:generateContent?key=${apiKey}`;

                    const response = await fetch(apiUrl, {
                        method: 'POST',
                        headers: { 'Content-Type': 'application/json' },
                        body: JSON.stringify(payload)
                    });

                    if (!response.ok) {
                        const errorData = await response.json();
                        throw new Error(`API error: ${errorData.error.message || response.statusText}`);
                    }

                    const result = await response.json();

                    if (result.candidates && result.candidates.length > 0 &&
                        result.candidates[0].content && result.candidates[0].content.parts &&
                        result.candidates[0].content.parts.length > 0) {
                        const text = result.candidates[0].content.parts[0].text;
                        interrogationOutput.innerHTML = `<p>${text.replace(/\n/g, '<br>')}</p>`;
                    } else {
                        displayInterrogationError('Could not generate a response. Please try again.');
                        interrogationOutput.innerHTML = '<p class="text-gray-400">Coach Bleak is ready for your questions about this audit report.</p>';
                    }
                } catch (err) {
                    console.error('Error calling Gemini API for report interrogation:', err);
                    displayInterrogationError(`Failed to get response: ${err.message}. Please check your input and try again.`);
                    interrogationOutput.innerHTML = '<p class="text-gray-400">Coach Bleak is ready for your questions about this audit report.</p>';
                } finally {
                    setInterrogationLoading('askReportBtn', false);
                    setInterrogationLoading('extendScenarioBtn', false);
                }
            }

            // Event listeners for Report Interrogation buttons
            askReportBtn.addEventListener('click', () => {
                const query = interrogationInput.value.trim();
                if (query) {
                    const prompt = `Answer the following question based *only* on the provided audit report content: "${query}"`;
