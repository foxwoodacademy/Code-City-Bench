        // Function to generate Chart.js Radar Chart for Corruption Probability Matrix
        function generateCorruptionMatrix(data, explanation) {
            corruptionMatrixPanel.classList.remove('hidden');
            matrixExplanation.textContent = explanation;

            if (corruptionMatrixChartInstance) {
                corruptionMatrixChartInstance.destroy();
            }

            const ctx = corruptionMatrixChartCanvas.getContext('2d');
            corruptionMatrixChartInstance = new Chart(ctx, {
                type: 'radar',
                data: {
                    labels: ['Legal Risk', 'Public Harm', 'Cover-Up Likelihood', 'Whistleblower Retaliation Risk', 'Financial Exposure'],
                    datasets: [{
                        label: 'Scenario Risk Score',
                        data: data,
                        backgroundColor: 'rgba(59, 130, 246, 0.4)',
                        borderColor: 'rgba(59, 130, 246, 1)',
                        borderWidth: 2,
                        pointBackgroundColor: 'rgba(59, 130, 246, 1)',
                        pointBorderColor: '#fff',
                        pointHoverBackgroundColor: '#fff',
                        pointHoverBorderColor: 'rgba(59, 130, 246, 1)'
                    }]
                },
                options: {
                    responsive: true,
                    maintainAspectRatio: false,
                    scales: {
                        r: {
                            angleLines: {
                                color: '#4a5568'
                            },
                            grid: {
                                color: '#4a5568'
                            },
                            pointLabels: {
                                color: '#e2e8f0',
                                font: {
                                    size: 14
                                },
                                callback: function(value) {
                                    // Wrap labels if too long
                                    const maxLength = 16;
                                    if (value.length > maxLength) {
                                        return value.split(' ').map((word, index, arr) => {
                                            let currentLine = word;
                                            if (index > 0 && (currentLine.length + arr[index - 1].length) > maxLength) {
                                                return '\n' + word;
                                            }
                                            return word;
                                        }).join(' ');
                                    }
                                    return value;
                                }
                            },
                            ticks: {
                                display: false,
                                beginAtZero: true,
                                max: 10
                            }
                        }
                    },
                    plugins: {
                        legend: {
                            display: false
                        },
                        tooltip: {
                            callbacks: {
                                label: function(context) {
                                    return context.dataset.label + ': ' + context.raw;
                                }
                            }
                        }
                    }
                }
            });
        }

        // Function to simulate AI analysis and response
        async function simulateAnalysis(followUpQuestion = '') {
            const input = scenarioInput.value.trim();
            if (!input) {
                outputArea.innerHTML = '<p class="text-red-400 text-center">Please describe a scenario to analyze.</p>';
                return;
            }

            setLoading(true);
            resetDynamicPanels(); // Clear all panels before new analysis
            outputArea.innerHTML = '<p class="text-blue-400 text-center animate-pulse">Analyzing patterns... Identifying vulnerabilities...</p>';
            exposedOverlay.classList.remove('active');
            puddleEffect.classList.remove('active');

            try {
                let prompt = '';
                let additionalInstructions = '';

                // Construct prompt based on mode
                if (currentMode === 'analysis') {
                    prompt = `As Coach Bleak, the Accountability Engine, provide a direct, uncompromising, and legally-oriented analysis of potential ethical breaches, legal ramifications, and the concept of accountability for the following alleged wrongdoing or corrupt scenario. Focus on the impact on the wrongdoer and the system, using strong, formal language. Do not provide legal advice, but highlight the principles of justice.`;
                    additionalInstructions = `\n\nScenario: "${input}"\n\nCoach Bleak's Analysis:`;
                } else if (currentMode === 'audit') {
                    prompt = `As Coach Bleak in AUDIT MODE, dissect the following scenario. Highlight potential discrepancies, financial anomalies, and hidden liabilities. Describe how a "Paper Trail Incinerator" would expose doctored documents and how an "Entity Spider" would map a "corruption web." Provide a "Corruption Probability Matrix" score (5 values, 1-10) and a "Liability Calculator" estimate (e.g., "Avg. sentence X yrs, Fines $Y").`;
                    additionalInstructions = `\n\nScenario: "${input}"\n\nAudit Report:`;
                } else if (currentMode === 'psychoWarfare') {
                    prompt = `As Coach Bleak in PSYCHOLOGICAL WARFARE MODE, analyze the following scenario. Generate a "Fear Index" score (1-100). Create a plausible "Echo Chamber" leak (e.g., "Internal memo suggests..."). Provide an "Unblinking Eye" count (e.g., "X eyes are here"). Give an "Ego Crusher" statement (reputation/status loss). Provide a "Last Seen Alive" update (e.g., "Target's LinkedIn was active X mins ago").`;
                    additionalInstructions = `\n\nScenario: "${input}"\n\nPsychological Impact Report:`;
                }

                if (followUpQuestion) {
                    prompt += `\n\nFollow-up Question: "${followUpQuestion}"`;
                }

                let chatHistory = [];
                chatHistory.push({ role: "user", parts: [{ text: prompt + additionalInstructions }] });

                const payload = { contents: chatHistory };
                const apiKey = ""; // Canvas will provide this key at runtime
                const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.0-flash:generateContent?key=${apiKey}`;

                const response = await fetch(apiUrl, {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify(payload)
                });

                if (!response.ok) {
                    const errorData = await response.json();
                    throw new Error(`API error: ${errorData.error.message || response.statusText}`);
                }

                const result = await response.json();

                let aiResponse = '';
                if (result.candidates && result.candidates.length > 0 &&
                    result.candidates[0].content && result.candidates[0].content.parts &&
                    result.candidates[0].content.parts.length > 0) {
                    aiResponse = result.candidates[0].content.parts[0].text;
                } else {
                    aiResponse = "Analysis inconclusive. Further data required. The truth remains obscured for now.";
                }

                outputArea.innerHTML = `<p class="text-gray-200">${aiResponse.replace(/\n/g, '<br>')}</p>`;

                // Trigger "exposed" visuals
                exposedOverlay.classList.add('active');
                puddleEffect.classList.add('active');

                // --- Process Mode-Specific Outputs ---
                if (currentMode === 'audit') {
                    consequencesPanel.classList.remove('hidden');
                    // Extract Liability Calculator and Chain of Consequences
                    const liabilityMatch = aiResponse.match(/Liability Calculator: (.+)/);
                    const chainMatch = aiResponse.match(/Chain of Consequences: (.+)/);
                    consequencesContent.innerHTML = `
                        <p class="mb-2"><strong>Liability Estimate:</strong> ${liabilityMatch ? liabilityMatch[1] : 'Not available.'}</p>
                        <p><strong>Chain of Consequences:</strong> ${chainMatch ? chainMatch[1] : 'Not available.'}</p>
                    `;

                    // Extract Corruption Probability Matrix data
                    const matrixMatch = aiResponse.match(/Corruption Probability Matrix: \[([^\]]+)\]/);
                    if (matrixMatch) {
                        const dataString = matrixMatch[1];
                        const dataValues = dataString.split(',').map(Number);
                        if (dataValues.length === 5 && dataValues.every(val => !isNaN(val))) {
                            generateCorruptionMatrix(dataValues, "This radar chart scores the scenario on key risk factors, providing a quick overview of its potential severity.");
                        } else {
