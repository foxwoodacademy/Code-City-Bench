import datetime

class FOIAGeneratorMicroAgent:

Dear [Agency],

I am writing to request access to the following records under the Freedom of Information Act (FOIA):

[Records Description]

I would like to request [Format] copies of the records. If the records are not available in the requested format, please provide them in an alternative format.

Please respond to this request within the statutory timeframe.

Sincerely,
[Your Name]
"""

    def generate_foia_request(self, agency: str, records_description: str, format: str, your_name: str):
        # Generate a FOIA request letter
        foia_request = self.template.replace("[Agency]", agency)
        foia_request = foia_request.replace("[Records Description]", records_description)
        foia_request = foia_request.replace("[Format]", format)
        foia_request = foia_request.replace("[Your Name]", your_name)
        foia_request = foia_request.replace("[Date]", datetime.date.today().strftime("%B %d, %Y"))
        return foia_request
