    <script>
        // Data for the illustrative Chart.js chart
        const completionData = {
            labels: [
                'Core Systems (100% Completed)',
                'Key AI Features (In Progress)',
                'UI/UX & Error Handling (Nearing Completion)',
                'Future Integrations (Early Prototype)'
            ],
            datasets: [{
                data: [50, 25, 15, 10], // Example percentages for illustration
                backgroundColor: [
                    '#06D6A0', // Green for completed
                    '#FFD166', // Yellow for in progress
                    '#118AB2', // Blue for nearing completion
                    '#FF6B6B'  // Red for early prototype
                ],
                hoverOffset: 4
            }]
        };

        // Function to wrap labels longer than 16 characters
        function wrapLabel(label) {
            if (label.length <= 16) {
                return label;
            }
            const words = label.split(' ');
            let lines = [];
            let currentLine = '';
            words.forEach(word => {
                if ((currentLine + word).length > 16 && currentLine !== '') {
                    lines.push(currentLine.trim());
                    currentLine = word + ' ';
                } else {
                    currentLine += word + ' ';
                }
            });
            lines.push(currentLine.trim());
            return lines;
        }

        // Apply label wrapping to the chart data
        completionData.labels = completionData.labels.map(wrapLabel);

        // Chart.js configuration
        const ctx = document.getElementById('projectCompletionChart').getContext('2d');
        const projectCompletionChart = new Chart(ctx, {
            type: 'doughnut',
            data: completionData,
            options: {
                responsive: true,
                maintainAspectRatio: false, // Crucial for responsive sizing within container
                plugins: {
                    legend: {
                        position: 'bottom',
                        labels: {
                            color: '#E0E0E0', // Light grey for legend text
                            font: {
                                size: 14
                            }
                        }
                    },
                    tooltip: {
                        callbacks: {
                            title: function(tooltipItems) {
                                const item = tooltipItems[0];
                                let label = item.chart.data.labels[item.dataIndex];
                                if (Array.isArray(label)) {
                                    return label.join(' ');
                                } else {
                                    return label;
                                }
                            },
                            label: function(context) {
                                let label = context.label || '';
                                if (Array.isArray(label)) {
                                    label = label.join(' '); // Join for tooltip display if wrapped
                                }
                                if (context.parsed !== null) {
                                    label += ': ' + context.parsed + '%';
                                }
                                return label;
                            }
                        },
                        backgroundColor: '#073B4C', // Dark blue tooltip background
                        titleColor: '#FFD166', // Yellow title in tooltip
                        bodyColor: '#E0E0E0', // Light grey body in tooltip
                        borderColor: '#118AB2',
                        borderWidth: 1
                    }
                }
            }
        });

        // Plotly.js is loaded via CDN but not explicitly used for a chart in this specific infographic
        // because the data from the source material is best represented by progress bars, flow diagrams,
        // and a simple Chart.js donut chart. If more complex statistical distributions or 3D data
        // were present, Plotly.js would be utilized, configured for Canvas/WebGL rendering.
    </script>
